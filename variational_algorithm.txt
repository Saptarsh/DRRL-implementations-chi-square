\section{Robust Bellman Equation}
\label{sec:model}

We consider a finite-state, finite-action, infinite-horizon discounted MDP
$(\cS,\cA,P_0,r,\gamma)$, where $P_0$ is the nominal transition kernel,
$r:\cS\times\cA\to[-1,1]$, and $\gamma\in(0,1)$.  For a policy $\pi$ and
transition kernel $P$, the standard Q-function is defined as
\[
Q_P^\pi(s,a)
=
\E\!\left[
\sum_{t=0}^\infty \gamma^t r(S_t,A_t)
\mid
S_0=s,A_0=a
\right].
\]
where the expectation is taken over the Markov chain defined by the following law: $\;
A_t\sim\pi(\cdot\mid S_t),\;
S_{t+1}\sim P(\cdot\mid S_t,A_t)$.
\subsection{Robust MDPs and the Uncertainty Set}

For each state-action pair $i=(s,a)$, define the chi-square uncertainty set
with uncertainty radius $\delta>0$ as
\begin{equation}
\label{eq:chi-set}
\cP_i^\chi
:=
\left\{
q\in\Delta_{\cS}:\;
q\ll P_0(\cdot\mid i),\;
D_\chi(q\|P_0(\cdot\mid i))\le \delta
\right\},
\end{equation}
where, for probability vectors $q,p\in\Delta_{\cS}$ with $q\ll p$,
\[
D_\chi(q\|p)
:=
\sum_{s'\in\cS}\frac{(q(s')-p(s'))^2}{p(s')}
=
\sum_{s'\in\cS}\frac{q(s')^2-p(s')^2}{p(s')}.
\]
We adopt the standard $(s,a)$-rectangular uncertainty model
(\cite{iyengar2005robust,nilim2005robust}), under which the uncertainty set is defined as
\begin{equation}
    \label{eq:uncertainty}
    \cP^\chi
=
\bigotimes_{i\in\cS\times\cA}\cP_i^\chi .
\end{equation}


For a value vector $V$, set
\begin{equation}
\sigma_i^\chi(V)=\inf_{q\in\cP_i^\chi}q^\top V.
\end{equation}
For a Q-function $Q$, let $V_Q(s)=\max_aQ(s,a)$.  The robust optimality operator is
\begin{equation}
(\mathcal T_\chi Q)(i)=r(i)+\gamma\sigma_i^\chi(V_Q).
\label{eq:robust-bellman}
\end{equation}
Under rectangularity, $\mathcal T_\chi$ is a $\gamma$-contraction in
$\ell_\infty$ (\cite{iyengar2005robust}) and hence has a unique fixed point.

Let $c_\delta=\sqrt{1+\delta}$.  The chi-square conditional value has the scalar dual \cite{duchi2019variance,duchi2021learning}
\begin{align}
\sigma_i^\chi(V)
&=\sup_{\eta\in\R}F_V^\chi(i;\eta),
\label{eq:chi-dual}\\
F_V^\chi(i;\eta)
&=\eta-c_\delta\sqrt{Z_V(i;\eta)},
\label{eq:scalar-dual-objective}\\
Z_V(i;\eta)
&=\E_{S'\sim P_0(\cdot\mid i)}[\pos{\eta-V(S')}^2\mid i].
\end{align}

Lemma~(\ref{lem:eta-bound}) shows that the supremum of the dual objective is achieved at a finite $\eta$.

\begin{lemma}[A compact dual interval]
\label{lem:eta-bound}
If $\norm V_\infty\le B$, then a maximizer in \eqref{eq:chi-dual} exists in $[-B,B_\eta]$, where
\begin{equation}
B_\eta=\frac{c_\delta+1}{c_\delta-1}B.
\label{eq:eta-bound}
\end{equation}
\end{lemma}

\begin{proof}
For $\eta\le\min_sV(s)$, $F_V^\chi(i;\eta)=\eta$, so no point below $-B$ is optimal.  For $\eta\ge B$, Jensen's inequality gives
$\sqrt{\E[(\eta-V(S'))^2\mid i]}\ge \eta-\E[V(S')\mid i]\ge\eta-B$.
Consequently $F_V^\chi(i;\eta)\le c_\delta B-(c_\delta-1)\eta<-B$ whenever $\eta>B_\eta$, whereas $F_V^\chi(i;\min V)=\min V\ge-B$.  Continuity on the remaining compact interval proves attainment.
\end{proof}

\section{A Variational Objective}
\label{sec:variational}

For $z\ge0$, define $q_z(u)=z/(2u)+u/2$ on $u>0$.  Arithmetic--geometric mean, or direct differentiation, gives
\begin{equation}
\sqrt z=\inf_{u>0}q_z(u),
\quad
\argmin_{u>0}q_z(u)=\{\sqrt z\}\quad(z>0).
\label{eq:qol}
\end{equation}
Hence \eqref{eq:chi-dual} is equivalently
\begin{equation}
\sigma_i^\chi(V)=
\sup_{\eta\in\R,u>0}
\left\{
\eta-\frac{c_\delta}{2}
\left(\frac{Z_V(i;\eta)}{u}+u\right)
\right\}.
\label{eq:variational-dual}
\end{equation}
The numerator in \eqref{eq:variational-dual} is linear in the conditional second moment $Z_V$, not its square root.  This enables an unbiased estimation of the objective from a single successor data point from the nominal trajectory generated by a behavioral policy.

\subsection{Floor bias and Auxiliary Approximation Class}

The variational objective in Equation \eqref{eq:variational-dual} contains a
positive auxiliary variable. In our robust learning algorithm presented in Section~(\ref{sec:algorithm}), we use a gradient-based optimization over the variational objective. To stabilize the gradients, we restrict this variable to be at least
$\ell\in(0,1]$, that is, $u \geq \ell >0$. The following lemma shows that this restriction introduces a
uniformly controlled surrogate bias of at most $c_\delta\ell/2$.


\begin{lemma}[Lower-floor bias]
\label{lem:floor-bias}
Assume $s \geq 0$ be a real number. The minimizer of $q_{s^2}$ over $[\ell,\infty)$ is
\(
m_\ell(s):=\max\{\ell,s\}.
\)
Moreover,
\begin{equation}
\label{eq:floor-bias}
q_{s^2}(m_\ell(s))-s
=
\frac{(\ell-s)_+^2}{2\ell}
\le\frac\ell2.
\end{equation}

\end{lemma}
\begin{proof}
The unconstrained minimizer is $s$. If $s<\ell$, convexity makes the endpoint $\ell$ optimal, and
\[
q_{s^2}(\ell)-s
=
\frac{s^2}{2\ell}+\frac\ell2-s
=
\frac{(\ell-s)^2}{2\ell}.
\]
Since $0\le s<\ell$, the final quantity is at most $\ell/2$. If $s \geq \ell$, the bias is zero.
\end{proof}






Let $\psi(i)\in\mathbb R^{d_\nu}$ be a signed feature vector used to
approximate the scalar dual function:
\[
\eta(i)\approx\eta_\nu(i):=\psi(i)^\top\nu,
\qquad \nu\in\mathbb R^{d_\nu}.
\]

For fixed $V$ and $\nu$, the floor-constrained optimal auxiliary variable for each $i$ is
\begin{equation}
u^*_{\ell,V,\nu}(i)
=
\max\left\{
\ell,
\sqrt{
\mathbb E\!\left[
  \bigl(\psi(i)^\top\nu-V(S')\bigr)_+^2
  \mid i
\right]}
\right\}.
\label{eq:optimal-scale-map}
\end{equation}
Thus, the auxiliary approximation class must represent a nonnegative
state--action-dependent function with a lower floor $\ell$. At the same time, to retain the variational objective jointly concave on the learnable parameters, the approximation function should depend
affinely on its learnable parameter (see the proof of Proposition \ref{prop:concavity}). We use the following auxiliary scale representation for our theoretical results.

{\color{red}Should we call these feature $\chi_i$?}

Let $\chi(i)\in\mathbb R^{d_\chi}$ be a signed feature vector and define
\begin{equation}\begin{aligned}
   & X_i:=\chi(i)\chi(i)^\top,\qquad
g_\Theta(i):=\inner{\Theta}{X_i}_F
             =\chi(i)^\top\Theta\chi(i),
\\
& u_{\Theta,\ell}(i):=\ell+g_\Theta(i),
\label{eq:function-classes-aux}
\end{aligned}
\end{equation}
where $\Theta\in\mathbb S_+^{d_\chi}$ is a learnable positive-semidefinite
matrix. This representation guarantees $g_\Theta(i)\geq0$ for arbitrary
signed features, while remaining affine in $\Theta$.
Given one observed transition $(i,S')$ and a frozen value target $V$, define
\[
x_{V,\nu}(i,S'):=\bigl(\psi(i)^\top\nu-V(S')\bigr)_+.
\]
The one-transition variational objective is
\begin{equation}
\widehat\varphi_{V,\ell}(\nu,\Theta;i,S')
=
\psi(i)^\top\nu
-
\frac{c_\delta}{2}
\left[
\frac{x_{V,\nu}(i,S')^2}{u_{\Theta,\ell}(i)}
+
u_{\Theta,\ell}(i)
\right].
\label{eq:sample-objective}
\end{equation}

The following proposition establishes the joint concavity of the variational objective on the learnable parameters and derives the gradient expressions.
\begin{proposition}[Joint concavity and gradient expression]
\label{prop:concavity}
The sample objective in \eqref{eq:sample-objective} is jointly concave and
continuously differentiable in $(\nu,\Theta)$ on
$\mathbb R^{d_\nu}\times\mathbb S_+^{d_\chi}$. With
\[
x=\bigl(\psi(i)^\top\nu-V(S')\bigr)_+,
\quad
u=\ell+\inner{\Theta}{X_i}_F,
\]
its gradients are
\begin{equation}
\nabla_\nu\widehat\varphi_{V,\ell}
=
\left(1-c_\delta\frac{x}{u}\right)\psi(i),
\quad
\nabla_\Theta\widehat\varphi_{V,\ell}
=
\frac{c_\delta}{2}
\left(\frac{x^2}{u^2}-1\right)X_i.
\label{eq:sample-gradients}
\end{equation}
Moreover, when $\norm{\psi(i)}_2,\norm{\chi(i)}_2 < \infty$, these are conditionally unbiased gradients of
the corresponding population objective for the visited index $i$.
\end{proposition}

\begin{proof}
The quadratic-over-linear perspective
\[
(a,u)\longmapsto \frac{(a_+)^2}{u},
\qquad u>0,
\]
is jointly convex because
\[
\frac{(a_+)^2}{u}
=
\sup_{v\geq0}\{2va-v^2u\},
\]
which is a pointwise supremum of affine functions of $(a,u)$. Here,
$a=\psi(i)^\top\nu-V(S')$ and
$u=\ell+\inner{\Theta}{X_i}_F$ are affine in $(\nu,\Theta)$, and
$u\geq\ell>0$. Therefore, the negative quadratic-over-linear term is jointly
concave. The remaining terms are affine, proving joint concavity.

The function $a\mapsto(a_+)^2$ is continuously differentiable, including at
$a=0$. Direct differentiation gives \eqref{eq:sample-gradients}. Conditional
unbiasedness follows by taking expectation over
$S'\sim P_i^0$ and interchanging expectation and differentiation.
\end{proof}


\begin{remark} Differentiating the scalar $\chi^2$ dual directly produces a ratio of two
conditional moments. A one-transition plug-in estimate of this ratio is
generally biased and can be unstable when its denominator is small. Liang
et al. \cite{liang2024single} proposed an algorithm based on this moment-based gradient by tracking the
conditional moments on faster time scales before updating the dual variable in a slow time scale.
In contrast, the variational formulation yields gradients that are conditionally unbiased for the
visited state--action pair. The shared dual and auxiliary-scale parameters can
therefore be updated jointly on a single time scale. The constraint $u\geq\ell$ controls the only denominator, with the
resulting floor bias quantified explicitly in Lemma~\ref{lem:floor-bias}.
Thus, the dual-learning analysis can be based on projected stochastic
concave optimization rather than on tracking multiple moving moment critics.
\end{remark}
\begin{remark}[Auxiliary function class choice]
\label{rem:psd-motivation}
 The affine positive semi-definite (PSD) matrix representation for the auxiliary variables permits
general signed features $\chi_i$ while simultaneously enforcing positivity and
retaining a convex parameter domain. When entry-wise nonnegative auxiliary features are available, the cheaper vector
model
\[
g_\rho(i)=\chi(i)^\top\rho,
\qquad \chi(i)\geq0,\quad \rho\geq0,
\]
has the same positivity and affine-parameter properties, and the theoretical results of this paper can be adapted to such function class straightforwardly. The PSD model avoids
the nonnegative-feature restriction at the cost of more computational requirement compared to the above function class as discussed in Section~(\ref{sec:computation}). In practical implementations, one may instead enforce positivity through,
for example, the square or softplus of a neural-network output. Such
parameterizations need not preserve concavity in the network weights and are
therefore not covered by the convex finite-time analysis below.
\end{remark}


\section{Function Approximation and Assumptions}
\label{sec:assumptions}

We assume we have access to a behavior policy $\pi_b$ generating transition data in the nominal transition kernel. The generated trajectory is denoted by Markov chain $i_k=(S_k,A_k)$. For any vector indexed by state--action pairs, write
$\norm{x}_D=(x^\top D x)^{1/2}$. Let $\mathcal F_k$ be the pre-transition sigma-field containing the trajectory through $i_k$ and all algorithmic iterates computed before drawing $S_{k+1}$. 
We make the following assumptions which are standard in both non-robust and robust RL
analysis (\cite{srikant2019finite,li2022first,chen2023target,zhou2023natural}).

\begin{assumption}[Behavior-chain mixing]
\label{ass:mixing}
The Markov chain $i_k=(S_k,A_k)$ induced by the behavior policy
$\pi_b$ and the nominal transition kernel $P_0$ is irreducible and
aperiodic, with stationary distribution $d$; let $D=\diag(d)$.
Moreover, there exist constants $C_{\rm mix}<\infty$ and
$\rho_{\rm mix}\in(0,1)$ such that
\begin{equation}
\sup_{i\in\mathcal S\times\mathcal A}
\norm{P_b^L(i,\cdot)-d}_{\rm TV}
\leq C_{\rm mix}\rho_{\rm mix}^L,
\qquad L\geq0,
\label{eq:mixing}
\end{equation}
where $P_b$ denotes the transition kernel of the state--action chain
under $(P_0,\pi_b)$.
\end{assumption}

By the Markov property and the definition of the pre-transition
filtration $\mathcal F_k$, Assumption~\ref{ass:mixing} implies that,
almost surely, for every $k\geq L\geq0$,
\begin{equation}
\norm{
\Law(i_k\mid\mathcal F_{k-L})-d
}_{\rm TV}
\leq C_{\rm mix}\rho_{\rm mix}^L.
\label{eq:conditional-mixing}
\end{equation}
Consequently, for every bounded function
$f:\mathcal S\times\mathcal A\to\mathbb R$ that is measurable with respect to the lagged filtration $\mathcal F_{k-L}$, 
\begin{equation}
\left|
\E[f(i_k)\mid\mathcal F_{k-L}]
-\E_{i\sim d}[f(i)]
\right|
\leq
2\norm{f}_\infty C_{\rm mix}\rho_{\rm mix}^L.
\label{eq:function-mixing}
\end{equation}

The constants $C_{\mathrm{mix}}, \rho_{\mathrm{mix}}$ depend on the behavior policy and nominal kernel.


We approximate the robust $Q$-function as 
\[
Q_\theta(i) = \phi(i)^\top\theta. 
\]
where, the feature vector for stat-action pair $i$ is denoted by $\phi(i)\in\R^{d_\theta}$ and the learable parameter vector is $\theta\in\R^{d_\theta}$. Let $\Phi$ have row $i$ equal
to $\phi(i)^\top$; similarly, let $\Psi$ have row $i$ equal to
$\psi(i)^\top$.   We make the following assumption on the features:

\begin{assumption}[Feature Conditioning]
    \label{ass:features}



\begin{equation}
\norm{\phi(i)}_2,\norm{\psi(i)}_2,\norm{\chi(i)}_2\le1,
\quad
\Phi^\top D\Phi\succeq\mu_\phi I
\label{eq:feature-assumption}
\end{equation}
for $\mu_\phi>0$. 

\end{assumption}

 Let $B_Q=(1-\gamma)^{-1}$  and let $\clip$ denotes componentwise clipping onto $[-B_Q,B_Q]$.  Also, let
$\Pi_\phi$ be the $D$-orthogonal projection onto $\mathrm{range}(\Phi)$:
\begin{equation}
\cQ=\{\clip(\Phi\theta):\theta\in\R^{d_\theta}\},
\qquad
\Pi_\phi f=\Phi(\Phi^\top D\Phi)^{-1}\Phi^\top Df.
\end{equation}
The Q-function approximation error is
\begin{equation}
\mathcal E_Q=
\sup_{Q\in\cQ}
\norm{\clip(\Pi_\phi\mathcal T_\chi Q)-\mathcal T_\chi Q}_\infty.
\label{eq:Q-approx}
\end{equation}
  %No covariance lower bound is required for the dual or scale features because Stage~1 is analyzed directly by online concave regret rather than by parameter identification.

We optimize the dual and the auxiliary parameters over compact convex sets
\begin{align}
\mathcal M_\nu&=\{\nu:\norm\nu_2\le R_\nu\},\\
\mathcal M_\Theta&=\{\Theta\in\Sym^{d_\chi}:\Theta\succeq0,
\ \norm\Theta_F\le R_\Theta\},
\label{eq:parameter-sets}
\end{align}
{\color{blue}where $R_\nu$ is chosen large enough for the desired dual class.  Lemma~\ref{lem:eta-bound} gives a sufficient scalar range; failure of a shared $\nu$ to represent the optimal dual optimizers for all $i$ in a point-wise manner is retained explicitly as approximation error.}  Set
\begin{equation}
H=R_\nu+B_Q,
\quad
R_{\mathcal W}=\sqrt{R_\nu^2+R_\Theta^2},
\quad
\mathcal W=\mathcal M_\nu\times\mathcal M_\Theta.
\end{equation}
Then $0\le x\le H$, $\norm{X_i}_F\le1$, and
\begin{equation}
\ell\le u_{\Theta,\ell}(i)\le\ell+R_\Theta.
\label{eq:scale-range}
\end{equation}

For a frozen $V$, define the stationary population objective
\begin{equation}
J_{V,\ell}(\nu,\Theta)
=\sum_i d_i\E[\widehat\varphi_{V,\ell}(\nu,\Theta;i,S')\mid i]
\label{eq:population-objective}
\end{equation}
and $J_{V,\ell}^{\rm PSD,*}=\max_{(\nu,\Theta)\in\mathcal W}J_{V,\ell}(\nu,\Theta)$.  The structured dual/scale error used in the theoretical results is
\begin{equation}
\mathcal E_{\rm str}(\ell)=
\sup_{Q\in\cQ}
\left\{
\sum_i d_i\sigma_i^\chi(V_Q)-J_{V_Q,\ell}^{\rm PSD,*}
\right\}.
\label{eq:structured-error}
\end{equation}
Section~\ref{sec:approximation} decomposes the above error into more interpretable terms.

Proposition~\ref{prop:concavity} and the preceding bounds imply
\begin{align}
&\norm{(\nabla_\nu\widehat\varphi_{V,\ell},\nabla_\Theta\widehat\varphi_{V,\ell})}\le G_\ell,
\label{eq:Gell}\\
&G_\ell=\left[
\left(1+\frac{c_\delta H}{\ell}\right)^2
+\frac{c_\delta^2}{4}\left(1+\frac{H^2}{\ell^2}\right)^2
\right]^{1/2},\\
&|\widehat\varphi_{V,\ell}|\le B_\varphi(\ell)
\coloneqq R_\nu+\frac{c_\delta}{2}
\left(\frac{H^2}{\ell}+\ell+R_\Theta\right).
\label{eq:Bphi}
\end{align}
For fixed problem parameters, $G_\ell=O(\ell^{-2})$ and $B_\varphi(\ell)=O(\ell^{-1})$.

\section{Two-Stage Target-Network Algorithm}
\label{sec:algorithm}

Let \(I_n=(S_n,A_n)\) denote the state--action Markov chain generated by
the behavior policy \(\pi_b\) and the nominal kernel \(P_0\).  The algorithm
maintains one global trajectory cursor and never restarts this chain at a
stage or block boundary.  At the beginning of outer block \(t\), it freezes
\begin{equation}
\widehat Q_t=\clip(\Phi\widehat\theta_t),
\qquad
V_t(s)=\max_{a\in\cA}\widehat Q_t(s,a),
\label{eq:frozen-target}
\end{equation}
and approximately computes the projected robust Bellman update
\begin{equation}
\Phi\theta\approx\Pi_\phi\mathcal T_\chi\widehat Q_t.
\label{eq:frozen-target-equation}
\end{equation}
Stage~1 applies projected stochastic ascent to the one-transition
variational objective in \eqref{eq:sample-objective}.  Its constant stepsize
\(\alpha_N\) is selected for the prescribed Stage~1 length \(N\), and its
output is the average of the iterates at which the \(N\) sample gradients
are evaluated.  Thus the terminal iterate
\((\nu_{t,N},\Theta_{t,N})\) is not included in this average.  Stage~2
freezes these averages and performs linear
stochastic approximation with the decreasing stepsize
\(\beta_m=\beta_0/(m+h_Q)\).  At the end of the block, its final iterate
becomes the next target-network parameter.

\begin{algorithm}[t]
\caption{Variational chi-square robust Q-learning with a PSD scale}
\label{alg:main}
\footnotesize
\begin{algorithmic}[1]
\STATE \textbf{Input:} outer-block count \(T\), stage lengths \(N,K\),
floor \(\ell\in(0,1]\), stepsize parameters
\(\alpha_N,\beta_0,h_Q\), and behavior policy \(\pi_b\).
\STATE Initialize \(\widehat\theta_0\) and the trajectory cursor \(n\gets0\).
\FOR{\(t=0,1,\ldots,T-1\)}
    \STATE Set \(\widehat Q_t=\clip(\Phi\widehat\theta_t)\) and
    \(V_t(s)=\max_{a\in\cA}\widehat Q_t(s,a)\).
    \STATE \textit{Stage 1:} initialize
    \((\nu_{t,0},\Theta_{t,0})=(0,0)\).
    \FOR{\(k=0,1,\ldots,N-1\)}
        \STATE Read the next transition
        \((i,s')\gets(I_n,S_{n+1})\).
        \STATE Compute
        \(x_{t,k}=\pos{\psi(i)^\top\nu_{t,k}-V_t(s')}\) and
        \(u_{t,k}=\ell+\inner{\Theta_{t,k}}{X_i}_F\).
        \STATE \(g^\nu_{t,k}\gets
        \left(1-c_\delta\frac{x_{t,k}}{u_{t,k}}\right)\psi(i).
        \)
        \STATE \(g^\Theta_{t,k}\gets
        \frac{c_\delta}{2}
        \left(\frac{x_{t,k}^2}{u_{t,k}^2}-1\right)X_i.\)
        \STATE \(\nu_{t,k+1}\gets
        \Proj_{\mathcal M_\nu}
        \bigl(\nu_{t,k}+\alpha_Ng^\nu_{t,k}\bigr).\)
        \STATE \(\Theta_{t,k+1}\gets
        \Proj_{\mathcal M_\Theta}
        \bigl(\Theta_{t,k}+\alpha_Ng^\Theta_{t,k}\bigr).\)
        \STATE \(n\gets n+1\).
    \ENDFOR
    \STATE \(\displaystyle
    \bar\nu_t\gets N^{-1}\sum_{k=0}^{N-1}\nu_{t,k}\), \quad
    \(\displaystyle
    \bar\Theta_t\gets N^{-1}\sum_{k=0}^{N-1}\Theta_{t,k}\).
    \STATE \textit{Stage 2:} initialize \(\theta_{t,0}=0\).
    \FOR{\(m=0,1,\ldots,K-1\)}
        \STATE Read the next transition
        \((i,s')\gets(I_n,S_{n+1})\).
        \STATE Compute
        \(\widetilde x_{t,m}=
        \pos{\psi(i)^\top\bar\nu_t-V_t(s')}\) and
        \(\widetilde u_{t,m}=
        \ell+\inner{\bar\Theta_t}{X_i}_F\).
        \STATE \(\displaystyle
        \widehat f_{t,m}\gets\psi(i)^\top\bar\nu_t-
        \frac{c_\delta}{2}
        \left(
        \frac{\widetilde x_{t,m}^2}{\widetilde u_{t,m}}
        +\widetilde u_{t,m}
        \right).\)
        \STATE \(Y_{t,m}\gets r(i)+\gamma\widehat f_{t,m}\), \quad
        \(\beta_m\gets\beta_0/(m+h_Q)\).
        \STATE \(\displaystyle
        \theta_{t,m+1}\gets\theta_{t,m}+
        \beta_m
        \bigl[Y_{t,m}-\phi(i)^\top\theta_{t,m}\bigr]\phi(i).\)
        \STATE \(n\gets n+1\).
    \ENDFOR
    \STATE \(\widehat\theta_{t+1}\gets\theta_{t,K}\).
\ENDFOR
\STATE \textbf{Output:} \(\widehat Q_T=\clip(\Phi\widehat\theta_T)\).
\end{algorithmic}
\end{algorithm}

For use in the analysis, define the global indices
\begin{equation}
b_t=t(N+K),
\qquad
q_t=b_t+N.
\label{eq:block-indices}
\end{equation}
Stage~1 of block \(t\) uses the transitions with indices
\(b_t,\ldots,q_t-1\), and its Q-stage uses those with indices
\(q_t,\ldots,b_{t+1}-1\).  Let
\begin{equation}
\mathcal B_t:=\mathcal F_{b_t},
\qquad
\mathcal H_{t,m}:=\mathcal F_{q_t+m}.
\label{eq:block-filtrations}
\end{equation}
Thus \(V_t,\bar\nu_t,\bar\Theta_t,\theta_{t,m}\), and
\(I_{q_t+m}\) are \(\mathcal H_{t,m}\)-measurable, whereas
\[
S_{q_t+m+1}\mid\mathcal H_{t,m}
\sim P_0(\cdot\mid I_{q_t+m}).
\]
Consequently,
\begin{equation}
\E[Y_{t,m}\mid\mathcal H_{t,m}]
=
\overline Y_{V_t,\bar\nu_t,\bar\Theta_t}(I_{q_t+m}),
\label{eq:q-conditional-mean}
\end{equation}
where
\begin{align}
\overline Y_{V,\nu,\Theta}(i)
&=
r(i)+\gamma f_{V,\ell}(i;\nu,\Theta),
\label{eq:mean-target}\\
f_{V,\ell}(i;\nu,\Theta)
&=
\psi(i)^\top\nu
\nonumber\\
&\quad
-
\frac{c_\delta}{2}
\left[
\frac{Z_V(i;\psi(i)^\top\nu)}
     {\ell+g_\Theta(i)}
+\ell+g_\Theta(i)
\right].
\label{eq:population-target}
\end{align}
The bounds in \eqref{eq:Bphi} imply, uniformly over all feasible frozen
inputs,
\begin{align}
|Y_{t,m}|,\ 
\left|\overline Y_{V_t,\bar\nu_t,\bar\Theta_t}
       (I_{q_t+m})\right|
\;&\le B_Y(\ell),
\nonumber\\
B_Y(\ell)&:=1+\gamma B_\varphi(\ell).
\label{eq:BY}
\end{align}

\section{Main Results}
\label{sec:main-results}

We first introduce the explicit quantities appearing in the theorem.
For \(n\ge1\), let
\begin{equation}
\Lambda_n:=1+\log(n+1)+\log^2(n+1).
\label{eq:Lambda}
\end{equation}
Define
\begin{align}
q_{\rm mix}
&:=\log(1/\rho_{\rm mix}),
\nonumber\\
\kappa_{\rm mix}
&:=
1+\frac{1+\log(C_{\rm mix}\vee1)}{q_{\rm mix}},
\label{eq:kappa-mix}
\end{align}
and set
\begin{align}
A_{\rm mix}&:=1+2\kappa_{\rm mix},
&
D_{\rm mix}&:=4(1+\kappa_{\rm mix}).
\label{eq:stage1-explicit-constants}
\end{align}
The Stage~1 error envelope is
\begin{equation}
\overline\varepsilon_{\rm var}(N,\ell)
:=
A_{\rm mix}R_{\mathcal W}G_\ell
\sqrt{\frac{\Lambda_N}{N}}
+
D_{\rm mix}B_\varphi(\ell)\frac{\Lambda_N}{N}.
\label{eq:stage1-envelope}
\end{equation}

For the Q-stage, define
\begin{align}
p_Q&:=2\mu_\phi\beta_0,
\nonumber\\
\Lambda^Q_{K,h_Q}
&:=
1+\log(K+h_Q)
\label{eq:q-rate-definitions}
\end{align}
and
\begin{equation}
\mathfrak R_Q(K)
:=
\frac{\Lambda^Q_{K,h_Q}}{K+h_Q}.
\label{eq:q-rate}
\end{equation}
The constant \(C_Q\) is given explicitly in
\eqref{eq:CQ-explicit} below; in particular, it is independent of
\(N,K,\ell,T,\gamma,\delta\), and the frozen target after the factor
\(B_Y(\ell)\) has been extracted.  Write
\begin{equation}
\varepsilon_Q(K,\ell)
:=
B_Y(\ell)\sqrt{C_Q\mathfrak R_Q(K)}.
\label{eq:eps-Q}
\end{equation}
Finally, let
\begin{equation}
\Gamma_T:=\sum_{j=0}^{T-1}\gamma^j
=\frac{1-\gamma^T}{1-\gamma},
\qquad
\Gamma_0:=0.
\label{eq:GammaT}
\end{equation}

\begin{theorem}[End-to-end convergence]
\label{thm:main}
Suppose Assumptions~\ref{ass:mixing} and~\ref{ass:features} and the
compact constraints in \eqref{eq:parameter-sets} hold.  In
Algorithm~\ref{alg:main}, take \(\widehat\theta_0\) deterministic and use
\begin{equation}
\alpha_N
=
\frac{R_{\mathcal W}}{G_\ell\sqrt{N\Lambda_N}},
\qquad
\beta_m=\frac{\beta_0}{m+h_Q},
\label{eq:main-stepsizes}
\end{equation}
where \(\beta_0>0\) is chosen so that
\begin{align}
p_Q&=2\mu_\phi\beta_0>1,
\label{eq:q-clean-gain}\\
h_Q&\ge\max\{1,2\beta_0\}.
\label{eq:q-implementable-offset}
\end{align}
Then, for all integers \(N,K\ge1\) and \(T\ge0\),
\begin{align}
&\E\norm{\widehat Q_T-Q_\chi^*}_\infty
\nonumber\\
&\le
\gamma^T\norm{\widehat Q_0-Q_\chi^*}_\infty
\nonumber\\
&\quad+
\Gamma_T
\Bigg\{
\mathcal E_Q
+\varepsilon_Q(K,\ell)
\nonumber\\
&\hspace{19mm}
+\frac{\gamma}{\mu_\phi}
\left[
\overline\varepsilon_{\rm var}(N,\ell)
+\mathcal E_{\rm str}(\ell)
\right]
\Bigg\}.
\label{eq:main-bound}
\end{align}
\end{theorem}

\begin{remark}[Gain choice and implementable offset]
\label{rem:q-offset}
Assumption~\ref{ass:features} implies
\[
\Phi^\top D\Phi
=\sum_i d_i\phi(i)\phi(i)^\top
\preceq I,
\]
and hence \(\mu_\phi\le1\).  Therefore the offset condition
\eqref{eq:q-implementable-offset} implies both
\(\beta_m\le1\) and
\(h_Q\ge2\mu_\phi\beta_0\), without requiring the algorithm to know
\(\mu_\phi\).  By contrast, the strict gain condition
\eqref{eq:q-clean-gain} is used only to
state and prove the principal result in its clean
\(\widetilde O(K^{-1/2})\) norm-rate regime; it is not required for the
updates to be well defined, stable, or convergent.  The cases
\(p_Q=1\) and \(0<p_Q<1\) are given in
Remark~\ref{rem:q-general-gain}.
If a known lower bound
\(\underline\mu_\phi\le\mu_\phi\) is available, choosing
\(\beta_0>(2\underline\mu_\phi)^{-1}\) guarantees \(p_Q>1\).
Without such a bound, \(\beta_0\) remains a tunable hyperparameter, and
the relaxed-rate guarantee applies for every \(\beta_0>0\).
As is typical at the boundary of a harmonic-stepsize recursion, the
displayed constant \(C_Q\) is not uniform as \(p_Q\downarrow1\); its
explicit expression contains \(1/(p_Q-1)\).
\end{remark}

\begin{corollary}[Explicit sample complexity]
\label{cor:complexity}
Under the conditions of Theorem~\ref{thm:main}, fix
\(c_\ell\in(0,1]\) and set
\(\ell=c_\ell\epsilon\) for \(\epsilon\in(0,1)\).  Define
\begin{align}
\overline G
&:=
1+c_\delta H+\frac{c_\delta}{2}(1+H^2),
\nonumber\\
\overline B_\varphi
&:=
R_\nu+\frac{c_\delta}{2}(H^2+1+R_\Theta),
\nonumber\\
\overline B_Y
&:=
1+\gamma\overline B_\varphi .
\label{eq:problem-envelopes}
\end{align}
Assume
\begin{equation}
\mathcal E_Q
+\frac{\gamma}{\mu_\phi}
\mathcal E_{\rm str}(c_\ell\epsilon)
\le\frac{(1-\gamma)\epsilon}{4}.
\label{eq:floor-condition}
\end{equation}
It is sufficient to choose \(N,K\) so that
\begin{align}
\frac{N}{\Lambda_N}
&\ge
\max\left\{
\left[
\frac{
8\gamma A_{\rm mix}R_{\mathcal W}\overline G
}{
\mu_\phi(1-\gamma)c_\ell^2
}
\right]^2\epsilon^{-6},
\right.
\nonumber\\[-1mm]
&\hspace{28mm}\left.
\frac{
8\gamma D_{\rm mix}\overline B_\varphi
}{
\mu_\phi(1-\gamma)c_\ell
}\epsilon^{-2}
\right\},
\label{eq:N-explicit-choice}\\
\frac{K+h_Q}{\Lambda^Q_{K,h_Q}}
&\ge
\frac{16C_Q\overline B_Y^2}
{c_\ell^2(1-\gamma)^2}\epsilon^{-4}.
\label{eq:K-explicit-choice}
\end{align}
With
\begin{equation}
B_0:=\norm{\widehat Q_0-Q_\chi^*}_\infty,
\qquad
T=
\left\lceil
\frac{\log((4B_0/\epsilon)\vee1)}
{\log(1/\gamma)}
\right\rceil,
\label{eq:block-count}
\end{equation}
Algorithm~\ref{alg:main} satisfies
\(\E\norm{\widehat Q_T-Q_\chi^*}_\infty\le\epsilon\).
For fixed problem, mixing, feature-class, and stepsize parameters, the
required trajectory length is therefore
\[
T(N+K)
=
\widetilde O\!\left(
\epsilon^{-6}\log(1/\epsilon)
\right).
\]
\end{corollary}

\begin{proof}
For \(0<\ell\le1\), the definitions in
\eqref{eq:Gell}--\eqref{eq:BY} imply
\begin{equation}
G_\ell\le\frac{\overline G}{\ell^2},
\qquad
B_\varphi(\ell)\le\frac{\overline B_\varphi}{\ell},
\qquad
B_Y(\ell)\le\frac{\overline B_Y}{\ell}.
\label{eq:envelope-inequalities}
\end{equation}
The two alternatives in \eqref{eq:N-explicit-choice} make the two
terms in
\(\gamma\overline\varepsilon_{\rm var}/
[\mu_\phi(1-\gamma)]\)
at most \(\epsilon/8\) each.
Equation~\eqref{eq:K-explicit-choice} makes
\(\varepsilon_Q/(1-\gamma)\le\epsilon/4\);
\eqref{eq:floor-condition} controls the combined approximation term;
and \eqref{eq:block-count} gives
\(\gamma^TB_0\le\epsilon/4\).
Theorem~\ref{thm:main} proves the claim.  Finally,
\(N=\widetilde O(\epsilon^{-6})\),
\(K=\widetilde O(\epsilon^{-4})\), and
\(T=O(\log(1/\epsilon))\) when the displayed problem parameters and
\(\gamma\) are fixed.
\end{proof}

\begin{remark}[Complexity under a relaxed gain]
\label{rem:slow-q}
At the boundary \(p_Q=1\), the extra logarithm in
Remark~\ref{rem:q-general-gain} leaves the same
\(\widetilde O(\epsilon^{-6})\) total sample-complexity order.  If
\(0<p_Q<1\), the Q-stage requirement becomes
\[
K=\widetilde O(\epsilon^{-4/p_Q})
\quad\text{when}\quad\ell=\Theta(\epsilon).
\]
Thus Stage~1 continues to determine the
\(\widetilde O(\epsilon^{-6})\) exponent whenever
\(p_Q\ge2/3\).  For \(p_Q<2/3\), the total exponent becomes
\(\max\{6,4/p_Q\}\).  This is the precise cost of removing the
clean-rate condition \(2\mu_\phi\beta_0>1\).
\end{remark}

\subsection{Greedy robust policy}

For a deterministic policy \(\pi\), define
\begin{equation}
(\mathcal T_\chi^\pi Q)(i)
=
r(i)+\gamma\sigma_i^\chi
\bigl(Q(\cdot,\pi(\cdot))\bigr),
\label{eq:policy-operator}
\end{equation}
and denote its unique fixed point by \(Q_\chi^\pi\).

\begin{corollary}[Greedy robust policy]
\label{cor:greedy}
Let \(\widehat\pi\) be greedy with respect to \(\widehat Q_T\).
Then
\begin{equation}
\norm{Q_\chi^{\widehat\pi}-Q_\chi^*}_\infty
\le
\frac{2\gamma}{1-\gamma}
\norm{\widehat Q_T-Q_\chi^*}_\infty.
\label{eq:policy-bound}
\end{equation}
\end{corollary}

\begin{proof}
Set \(e=\norm{\widehat Q_T-Q_\chi^*}_\infty\).
Greediness gives, for every \(s\),
\[
0\le
V_{Q_\chi^*}(s)-Q_\chi^*(s,\widehat\pi(s))
\le2e.
\]
Indeed, insert \(\widehat Q_T\) on both sides of its greedy inequality.
The map \(V\mapsto\sigma_i^\chi(V)\) is one-Lipschitz in
\(\ell_\infty\), and hence
\[
\norm{
\mathcal T_\chi Q_\chi^*
-\mathcal T_\chi^{\widehat\pi}Q_\chi^*
}_\infty
\le2\gamma e.
\]
Use
\(Q_\chi^*=\mathcal T_\chi Q_\chi^*\),
\(Q_\chi^{\widehat\pi}
=\mathcal T_\chi^{\widehat\pi}Q_\chi^{\widehat\pi}\),
and the \(\gamma\)-contraction of
\(\mathcal T_\chi^{\widehat\pi}\) to obtain
\[
\norm{Q_\chi^*-Q_\chi^{\widehat\pi}}_\infty
\le
2\gamma e
+\gamma\norm{Q_\chi^*-Q_\chi^{\widehat\pi}}_\infty.
\]
Rearrangement proves \eqref{eq:policy-bound}.
\end{proof}
